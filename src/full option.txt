#include <BleMouse.h>
#include <Adafruit_MPU6050.h>
#include "edge-impulse-sdk/classifier/ei_run_classifier.h"

#define LEFTBUTTON 26
#define RIGHTBUTTON 25
#define JOY_X 33 // URX
#define JOY_Y 32 // URY
#define SPEED 10

Adafruit_MPU6050 mpu;
BleMouse bleMouse;

bool sleepMPU = true;
bool isChivMode = false; // false = Air Mouse ปกติ, true = โหมด Chivalry 2

// ตัวแปรสำหรับเช็คการกดปุ่มค้างเพื่อสลับโหมด
unsigned long bothPressedMillis = 0;
bool isBothPressed = false;

// ตัวแปรสำหรับ Cooldown การโจมตี (ปรับลดเหลือ 200 มิลลิวินาที)
unsigned long lastActionTime = 0;
const unsigned long COOLDOWN_MS = 200; 

// ตัวแปรสำหรับจัดเก็บข้อมูลเซนเซอร์ส่งเข้า Edge Impulse
#define FREQUENCY_HZ 50
#define INTERVAL_MS (1000 / FREQUENCY_HZ)
unsigned long lastSampleTime = 0;
float feature_buffer[EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE];
int buffer_index = 0;

// ฟังก์ชันดึงข้อมูลดิบให้ Edge Impulse
int raw_feature_get_data(size_t offset, size_t length, float *out_ptr) {
    memcpy(out_ptr, feature_buffer + offset, length * sizeof(float));
    return 0;
}

void setup() {
    Serial.begin(115200);

    pinMode(LEFTBUTTON, INPUT_PULLUP);
    pinMode(RIGHTBUTTON, INPUT_PULLUP);

    bleMouse.begin();

    delay(1000);
    if (!mpu.begin()) {
        Serial.println("Failed to find MPU6050 chip");
        while (1) { delay(10); }
    }
    Serial.println("MPU6050 Found!");

    mpu.enableSleep(sleepMPU);
}

void loop() {
    if (bleMouse.isConnected()) {
        if (sleepMPU) {
            delay(3000);
            Serial.println("MPU6050 awakened!");
            sleepMPU = false;
            mpu.enableSleep(sleepMPU);
            delay(500);
        }

        // --- ระบบสลับโหมด: กดปุ่ม GPIO25 และ GPIO26 ค้างไว้ 2.5 วินาที ---
        bool leftState = !digitalRead(LEFTBUTTON);
        bool rightState = !digitalRead(RIGHTBUTTON);

        if (leftState && rightState) {
            if (!isBothPressed) {
                isBothPressed = true;
                bothPressedMillis = millis();
            } else {
                if (millis() - bothPressedMillis >= 2500) {
                    isChivMode = !isChivMode;
                    Serial.print("MODE SWITCHED! Chivalry 2 Mode: ");
                    Serial.println(isChivMode ? "ON (Combat)" : "OFF (Air Mouse)");
                    delay(1000);
                    isBothPressed = false;
                }
            }
        } else {
            isBothPressed = false;
        }

        // --- อ่านค่า MPU6050 ---
        sensors_event_t a, g, temp;
        mpu.getEvent(&a, &g, &temp);

        float moveX = g.gyro.z * SPEED;
        float moveY = g.gyro.y * -SPEED;

        // ==========================================
        // MODE 1: AIR MOUSE ปกติ (isChivMode = false)
        // ==========================================
        if (!isChivMode) {
            bleMouse.move(moveX, moveY);

            if (leftState && !isBothPressed) {
                bleMouse.click(MOUSE_LEFT);
                delay(40);
            }
            if (rightState && !isBothPressed) {
                bleMouse.click(MOUSE_RIGHT);
                delay(40);
            }
        }
        // ==========================================
        // MODE 2: CHIVALRY 2 COMBAT MODE (isChivMode = true)
        // ==========================================
        else {
            // 1. ควบคุมการหันมุมกล้องด้วย PS2 Joystick
            int joyX = analogRead(JOY_Y);
            int joyY = analogRead(JOY_X);
            
            int joyMoveX = 0;
            int joyMoveY = 0;
            
            if (joyX > 2300) {
                joyMoveX = -map(joyX, 2300, 4095, 1, 12);
            } else if (joyX < 1700) {
                joyMoveX = -map(joyX, 1700, 0, -1, -12);
            }
            
            if (joyY > 2300) {
                joyMoveY = -map(joyY, 2300, 4095, -1, -12);
            } else if (joyY < 1700) {
                joyMoveY = -map(joyY, 1700, 0, 1, 12);
            }
            
            bleMouse.move(joyMoveX, joyMoveY);

            // 2. ควบคุมการบล็อคด้วยปุ่ม RIGHTBUTTON
            if (leftState && !isBothPressed) {
                bleMouse.press(MOUSE_LEFT);
            } else {
                bleMouse.release(MOUSE_LEFT);
            }
            if (rightState && !isBothPressed) {
                bleMouse.press(MOUSE_RIGHT);
            } else {
                bleMouse.release(MOUSE_RIGHT);
            }

            // 3. เก็บข้อมูลท่าทางจาก MPU6050
            if (millis() - lastSampleTime >= INTERVAL_MS) {
                lastSampleTime = millis();

                if (buffer_index < EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE) {
                    feature_buffer[buffer_index++] = a.acceleration.x;
                    feature_buffer[buffer_index++] = a.acceleration.y;
                    feature_buffer[buffer_index++] = a.acceleration.z;
                    feature_buffer[buffer_index++] = g.gyro.x;
                    feature_buffer[buffer_index++] = g.gyro.y;
                    feature_buffer[buffer_index++] = g.gyro.z;
                }

                // เมื่อ Buffer เต็ม ให้รัน Classifier เพื่อทำนายท่าโจมตี
                if (buffer_index >= EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE) {
                    ei_signal_t signal;
                    signal.total_length = EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE;
                    signal.get_data = &raw_feature_get_data;

                    ei_impulse_result_t result = { 0 };
                    EI_IMPULSE_ERROR res = run_classifier(&signal, &result, false);

                    if (res == EI_IMPULSE_OK) {
                        int predicted_index = 0;
                        float max_value = 0;
                        for (size_t i = 0; i < EI_CLASSIFIER_LABEL_COUNT; i++) {
                            if (result.classification[i].value > max_value) {
                                max_value = result.classification[i].value;
                                predicted_index = i;
                            }
                        }

                        // ตรวจสอบความมั่นใจ และเช็ค Cooldown 200ms
                        if (max_value > 0.7 && (millis() - lastActionTime > COOLDOWN_MS)) {
                            String label = result.classification[predicted_index].label;

                            if (label == "slash2left") {
                                bleMouse.click(MOUSE_LEFT);           
                                lastActionTime = millis();            
                            }
                            else if (label == "slash2right") {
                                bleMouse.click(MOUSE_FORWARD);        
                                lastActionTime = millis();
                            }
                            else if (label == "overhead2left") {
                                bleMouse.move(0, 0, -1);              
                                lastActionTime = millis();
                            }
                            else if (label == "overhead2right") {
                                bleMouse.click(MOUSE_BACK);           
                                lastActionTime = millis();
                            }
                            else if (label == "stab") {
                                bleMouse.move(0, 0, 1);               
                                lastActionTime = millis();
                            }
                        }
                    }
                    
                    // สำคัญ: สำหรับท่าทางแบบพุ่งไปทีเดียวจบ ให้ล้าง Buffer ทั้งหมดทันที
                    // เพื่อให้รอบถัดไปเป็นการเก็บข้อมูลท่าใหม่จากจุดเริ่มต้นจริงๆ
                    buffer_index = 0;
                }
            }
        }
    }
}
